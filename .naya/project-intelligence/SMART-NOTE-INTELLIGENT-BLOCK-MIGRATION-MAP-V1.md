## MIGRATION EXECUTION SPEC

### Per-Artifact Transformation Steps

```python
def migrate_artifact(source_path: Path, ib_id: str) -> MigrationResult:
    """
    Transform a MIGRATE artifact to canonical IB format.
    """
    # 1. Read source content
    content = source_path.read_text(encoding='utf-8')
    
    # 2. Extract metadata
    metadata = extract_metadata(content, source_path)
    
    # 3. Generate canonical Smart Note
    smart_note = generate_canonical_smart_note(
        ib_id=ib_id,
        content=content,
        metadata=metadata,
        source_path=source_path
    )
    
    # 4. Extract machine layer
    machine_layer = extract_machine_layer(smart_note, metadata)
    
    # 5. Create target directory
    target_dir = Path(f"SUPERBRAIN/INTELLIGENCE/2026/09/25/{metadata.category}/{metadata.topic}/{ib_id}/")
    target_dir.mkdir(parents=True, exist_ok=True)
    
    # 5. Write canonical files
    (target_dir / "smart-note.md").write_text(smart_note)
    (target_dir / "machine-layer.json").write_text(json.dumps(machine_layer, indent=2))
    (target_dir / "provenance.json").write_text(json.dumps(build_provenance(source_path, ib_id), indent=2))
    
    # 6. Create evidence directory
    (target_dir / "evidence").mkdir(exist_ok=True)
    copy_evidence_files(source_path, target_dir / "evidence")
    
    # 7. Create empty projections directory
    (target_dir / "projections").mkdir(exist_ok=True)
    
    # 8. Validate
    lint_result = lint_canonical_smart_note(target_dir / "smart-note.md")
    if not lint_result.passed:
        return MigrationResult(success=False, errors=lint_result.errors)
    
    # 9. Archive original
    archive_original(source_path)
    
    return MigrationResult(success=True, ib_id=ib_id, target_path=target_dir)
```
