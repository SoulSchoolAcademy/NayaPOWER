The law, in CSS
.lv-board {
  background: linear-gradient(180deg, #121217, #070709); /* BLACK, not purple */
  border: 1px solid rgba(255,255,255,.10);
  border-radius: 22px; padding: 28px;
  /* glow stays AT the border — never bleeding out */
  box-shadow:
    inset 0 0 0 1px rgba(var(--lv-accent-rgb),.14),
    inset 0 1px 0 rgba(255,255,255,.09),
    0 0 30px -10px rgba(var(--lv-accent-rgb),.28),
    0 26px 60px rgba(0,0,0,.60);
}
.lv-board:hover {
  transform: translateY(-4px);              /* hover MOVES it */
  border-color: rgba(var(--lv-accent-rgb),.45);
  /* edge ignites, face lights up — light up, don't glow out */
}
